        const getWeekID = () => {
            const now = new Date();
            const d = new Date(Date.UTC(now.getFullYear(), now.getMonth(), now.getDate()));
            const dayNum = d.getUTCDay() || 7;
            d.setUTCDate(d.getUTCDate() + 4 - dayNum);
            const year = d.getUTCFullYear();
            const weekNo = Math.ceil(( ( (d - new Date(Date.UTC(year,0,1))) / 86400000) + 1)/7);
            return `${year}-W${weekNo}`;
        };

        const getWeekRange = () => {
            const curr = new Date();
            const first = curr.getDate() - curr.getDay();
            const last = first + 6;
            const firstday = new Date(curr.setDate(first)).toLocaleDateString('pt-BR');
            const lastday = new Date(curr.setDate(last)).toLocaleDateString('pt-BR');
            return `${firstday} até ${lastday}`;
        }

        function App() {
            const [view, setView] = useState('loading'); // loading, config, dashboard, inspection, admin
            const [user, setUser] = useState(null);
            const [selectedSetor, setSelectedSetor] = useState('');
            const [responsavel, setResponsavel] = useState('');
            
            // Dados Dinâmicos
            const [questions, setQuestions] = useState([]);
            const [answers, setAnswers] = useState({});
            const [obs, setObs] = useState({});
            const [inspectionsThisWeek, setInspectionsThisWeek] = useState([]);
            const [db, setDb] = useState(null);
            const [appId, setAppId] = useState('imac-app-v1'); // ID Fixo para o app do usuário

            // Admin States
            const [editingQ, setEditingQ] = useState(null);
            const [isModalOpen, setIsModalOpen] = useState(false);
            const [adminFilterSetor, setAdminFilterSetor] = useState('TODOS');

            // 1. Tentar conectar ao iniciar
            useEffect(() => {
                const tryConnect = async () => {
                    // Verifica se estamos no ambiente Canvas (tem variavel global)
                    let config = null;
                    let currentAppId = 'imac-app-v1';
                    
                    if (typeof __firebase_config !== 'undefined') {
                        // Ambiente Canvas (automático)
                        config = JSON.parse(__firebase_config);
                        if (typeof __app_id !== 'undefined') {
                            currentAppId = __app_id;
                        }
                    } else {
                        // Ambiente GitHub/Externo (Local Storage)
                        const stored = localStorage.getItem('imac_firebase_config');
                        if (stored) config = JSON.parse(stored);
                    }

                    setAppId(currentAppId);

                    if (config) {
                        await initFirebase(config, currentAppId);
                    } else {
                        setView('config');
                    }
                };
                
                // Aguarda scripts do modulo
                if (window.firebaseModules) {
                    tryConnect();
                } else {
                    window.addEventListener('firebase-ready', tryConnect);
                }
            }, []);

            const initFirebase = async (config, currentAppId) => {
                try {
                    const { initializeApp, getAuth, signInAnonymously, onAuthStateChanged, signInWithCustomToken, getFirestore } = window.firebaseModules;
                    
                    const app = initializeApp(config);
                    const auth = getAuth(app);
                    const firestore = getFirestore(app);
                    setDb(firestore);

                    // Tenta autenticação (Custom Token se estiver no Canvas, Anônimo se for externo)
                    if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                        await signInWithCustomToken(auth, __initial_auth_token);
                    } else {
                        await signInAnonymously(auth);
                    }
                    
                    onAuthStateChanged(auth, (u) => {
                        if (u) {
                            setUser(u);
                            loadData(firestore, currentAppId); 
                            setView('dashboard');
                        }
                    });
                } catch (err) {
                    console.error("Erro Conexão:", err);
                    // Se der erro (ex: config inválida), volta pra tela de config
                    if(typeof __firebase_config === 'undefined') {
                         localStorage.removeItem('imac_firebase_config');
                         setView('config');
                    }
                }
            };

            const loadData = (firestore, currentAppId) => {
                const { collection, query, where, onSnapshot } = window.firebaseModules;
                
                // Perguntas - Adicionando callback de erro
                const qQuestions = query(collection(firestore, 'artifacts', currentAppId, 'public', 'data', 'questions'));
                onSnapshot(qQuestions, (snapshot) => {
                    const data = [];
                    snapshot.forEach((doc) => data.push({ id: doc.id, ...doc.data() }));
                    setQuestions(data);
                }, (error) => {
                    console.warn("Erro carregando perguntas (pode estar vazio ou sem permissão):", error);
                });

                // Inspeções - Adicionando callback de erro
                const weekId = getWeekID();
                const qInspections = query(
                    collection(firestore, 'artifacts', currentAppId, 'public', 'data', 'inspections'),
                    where('weekId', '==', weekId)
                );
                onSnapshot(qInspections, (snapshot) => {
                    const data = [];
                    snapshot.forEach((doc) => data.push({ id: doc.id, ...doc.data() }));
                    setInspectionsThisWeek(data);
                }, (error) => {
                    console.warn("Erro carregando inspeções:", error);
                });
            };

            // --- LÓGICA DE CÁLCULO ---
            const allSetors = useMemo(() => {
                if (questions.length === 0) return [];
                const s = new Set(questions.map(p => p.setor).filter(x => x !== 'TODOS'));
                return Array.from(s).sort();
            }, [questions]);

            const setoresConcluidos = useMemo(() => new Set(inspectionsThisWeek.map(i => i.setor)), [inspectionsThisWeek]);

            const progress = useMemo(() => {
                const total = allSetors.length;
                const done = setoresConcluidos.size;
                const pct = total === 0 ? 0 : Math.round((done / total) * 100);
                return { total, done, pct };
            }, [allSetors, setoresConcluidos]);

            const currentQuestions = useMemo(() => {
                if (!selectedSetor) return [];
                return questions.filter(p => p.setor === selectedSetor || p.setor === 'TODOS');
            }, [selectedSetor, questions]);

            const scoreAtual = useMemo(() => {
                if (currentQuestions.length === 0) return 0;
                let total = 0;
                let conformes = 0;
                currentQuestions.forEach(q => {
                    if (answers[q.id]) { total++; if (answers[q.id] === 'C') conformes++; }
                });
                return total === 0 ? 0 : Math.round((conformes / total) * 100);
            }, [answers, currentQuestions]);

            // --- AÇÕES ---
            const handleStartInspection = (setor) => {
                if (!responsavel) { alert("Informe o responsável."); return; }
                setSelectedSetor(setor);
                setAnswers({});
                setObs({});
                setView('inspection');
            };

            const handleSaveInspection = async () => {
                if (!db) return;
                const { collection, addDoc, Timestamp } = window.firebaseModules;
                const pendentes = currentQuestions.filter(q => !answers[q.id]);
                if (pendentes.length > 0 && !confirm(`Existem ${pendentes.length} perguntas sem resposta. Salvar?`)) return;

                await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'inspections'), {
                    setor: selectedSetor,
                    responsavel: responsavel,
                    data: new Date().toISOString(),
                    timestamp: Timestamp.now(),
                    weekId: getWeekID(),
                    score: scoreAtual,
                    respostas: answers,
                    observacoes: obs,
                    status: 'Concluído'
                });
                setView('dashboard');
            };

            const handleImportDefaults = async () => {
                if (!confirm("Isso importará todas as perguntas padrão. Continuar?")) return;
                const { collection, writeBatch, doc } = window.firebaseModules;
                const batch = writeBatch(db);
                STATIC_DB_PERGUNTAS.forEach(q => {
                    const newRef = doc(collection(db, 'artifacts', appId, 'public', 'data', 'questions'));
                    batch.set(newRef, q);
                });
                await batch.commit();
                alert("Importação concluída!");
            };

            const handleSaveQuestion = async (e) => {
                e.preventDefault();
                const formData = new FormData(e.target);
                const qData = {
                    setor: formData.get('setor').toUpperCase(),
                    grupo: formData.get('grupo').toUpperCase(),
                    pergunta: formData.get('pergunta')
                };

                const { collection, addDoc, updateDoc, doc } = window.firebaseModules;
                if (editingQ && editingQ.id) {
                    await updateDoc(doc(db, 'artifacts', appId, 'public', 'data', 'questions', editingQ.id), qData);
                } else {
                    await addDoc(collection(db, 'artifacts', appId, 'public', 'data', 'questions'), qData);
                }
                setIsModalOpen(false);
                setEditingQ(null);
            };

            const handleDeleteQuestion = async (id) => {
                if (!confirm("Tem certeza que deseja excluir esta pergunta?")) return;
                const { deleteDoc, doc } = window.firebaseModules;
                await deleteDoc(doc(db, 'artifacts', appId, 'public', 'data', 'questions', id));
            };

            // --- RENDER ---
            
            if (view === 'loading') return <div className="flex h-screen bg-slate-50 items-center justify-center"><div className="animate-spin rounded-full h-12 w-12 border-b-2 border-blue-600"></div></div>;

            if (view === 'config') return <ConfigScreen onSave={(config) => initFirebase(config, appId)} />;

            if (view === 'dashboard') {
                return (
                    <div className="max-w-md mx-auto min-h-screen bg-slate-50 flex flex-col animate-fade-in">
                        <header className="bg-blue-900 text-white p-6 rounded-b-3xl shadow-lg mb-6 relative">
                            <button onClick={() => setView('admin')} className="absolute top-6 right-6 p-2 bg-white/10 rounded-full hover:bg-white/20">
                                <Icons.Gear />
                            </button>
                            <h1 className="text-2xl font-bold">Gestão IMAC</h1>
                            <p className="text-blue-200 text-sm mb-4">Controle de Qualidade</p>
                            
                            <div className="bg-white/10 backdrop-blur-md p-4 rounded-xl border border-white/20">
                                <div className="flex justify-between items-end mb-2">
                                    <span className="text-blue-100 text-sm">Meta Semanal ({getWeekRange()})</span>
                                    <span className="text-2xl font-bold text-white">{progress.pct}%</span>
                                </div>
                                <div className="w-full bg-blue-950/50 rounded-full h-3 mb-2">
                                    <div className="bg-yellow-400 h-3 rounded-full transition-all duration-1000" style={{ width: `${progress.pct}%` }}></div>
                                </div>
                                <p className="text-xs text-blue-200 text-right">{progress.done}/{progress.total} setores</p>
                            </div>
                        </header>

                        <main className="flex-1 px-6 pb-20">
                            <div className="mb-6">
                                <label className="block text-sm font-bold text-slate-700 mb-2">Responsável</label>
                                <input type="text" value={responsavel} onChange={(e) => setResponsavel(e.target.value)} placeholder="Digite seu nome..." className="w-full p-3 rounded-xl border border-slate-200 shadow-sm" />
                            </div>

                            {questions.length === 0 ? (
                                <div className="text-center py-10">
                                    <p className="text-slate-500 mb-4">Banco de dados limpo.</p>
                                    <button onClick={() => setView('admin')} className="text-blue-600 font-bold underline">Configurar no Admin</button>
                                </div>
                            ) : (
                                <div className="space-y-3">
                                    {allSetors.map(setor => {
                                        const isDone = setoresConcluidos.has(setor);
                                        const insp = inspectionsThisWeek.find(i => i.setor === setor);
                                        return (
                                            <button key={setor} onClick={() => handleStartInspection(setor)} className={`w-full p-4 rounded-xl flex items-center justify-between card-shadow transition-all active:scale-95 ${isDone ? 'bg-green-50 border border-green-200' : 'bg-white hover:bg-slate-50 border border-slate-100'}`}>
                                                <div className="flex items-center gap-3">
                                                    <div className={`h-10 w-10 rounded-full flex items-center justify-center ${isDone ? 'bg-green-100 text-green-600' : 'bg-slate-100 text-slate-400'}`}>
                                                        {isDone ? <Icons.Check /> : <Icons.List />}
                                                    </div>
                                                    <div className="text-left">
                                                        <p className={`font-bold ${isDone ? 'text-green-800' : 'text-slate-700'}`}>{setor}</p>
                                                        {isDone && <p className="text-xs text-green-600">Nota: {insp.score}%</p>}
                                                    </div>
                                                </div>
                                                {isDone ? <span className="text-xs bg-green-200 text-green-800 px-2 py-1 rounded font-bold">FEITO</span> : <span className="h-8 w-8 rounded-full bg-blue-50 text-blue-600 flex items-center justify-center">→</span>}
                                            </button>
                                        );
                                    })}
                                </div>
                            )}
                        </main>
                    </div>
                );
            }

            if (view === 'admin') {
                const filteredQuestions = adminFilterSetor === 'TODOS' 
                    ? questions 
                    : questions.filter(q => q.setor === adminFilterSetor || q.setor === 'TODOS');

                return (
                    <div className="max-w-md mx-auto min-h-screen bg-slate-50 flex flex-col animate-fade-in">
                        <header className="bg-slate-800 text-white p-4 flex items-center justify-between sticky top-0 z-10 shadow-md">
                            <button onClick={() => setView('dashboard')} className="text-sm font-bold text-slate-300">← Voltar</button>
                            <h1 className="font-bold">Administração</h1>
                            <div className="w-10"></div>
                        </header>

                        <main className="flex-1 p-4 pb-24">
                            <div className="flex gap-2 mb-4 overflow-x-auto pb-2 no-scrollbar">
                                <button onClick={() => setAdminFilterSetor('TODOS')} className={`whitespace-nowrap px-3 py-1 rounded-full text-xs font-bold ${adminFilterSetor === 'TODOS' ? 'bg-blue-600 text-white' : 'bg-white text-slate-600 border'}`}>Todos</button>
                                {allSetors.map(s => (
                                    <button key={s} onClick={() => setAdminFilterSetor(s)} className={`whitespace-nowrap px-3 py-1 rounded-full text-xs font-bold ${adminFilterSetor === s ? 'bg-blue-600 text-white' : 'bg-white text-slate-600 border'}`}>{s}</button>
                                ))}
                            </div>

                            {questions.length === 0 && (
                                <div className="bg-yellow-50 p-6 rounded-xl border border-yellow-200 text-center mb-6">
                                    <h3 className="font-bold text-yellow-800 mb-2">Importar Dados Iniciais</h3>
                                    <p className="text-xs text-yellow-700 mb-4">Seu banco está vazio. Clique para carregar as perguntas padrão da IMAC.</p>
                                    <button onClick={handleImportDefaults} className="bg-yellow-500 text-white px-4 py-2 rounded-lg font-bold shadow-md hover:bg-yellow-600 w-full">Importar Padrão</button>
                                </div>
                            )}
                            
                            {/* Botão Reset Config (Escondido no final, útil se errar a key) */}
                            <div className="mb-4 text-right">
                                <button onClick={()=>{localStorage.removeItem('imac_firebase_config'); window.location.reload()}} className="text-[10px] text-red-400 underline">Resetar Configuração Firebase</button>
                            </div>

                            <div className="space-y-3">
                                {filteredQuestions.map(q => (
                                    <div key={q.id} className="bg-white p-4 rounded-lg shadow-sm border border-slate-100 flex justify-between items-start">
                                        <div className="flex-1 pr-2">
                                            <div className="flex gap-2 mb-1">
                                                <span className="text-[10px] bg-slate-100 text-slate-500 px-1.5 py-0.5 rounded font-bold">{q.setor}</span>
                                                <span className="text-[10px] bg-blue-50 text-blue-600 px-1.5 py-0.5 rounded font-bold">{q.grupo}</span>
                                            </div>
                                            <p className="text-sm text-slate-800">{q.pergunta}</p>
                                        </div>
                                        <div className="flex flex-col gap-2">
                                            <button onClick={() => { setEditingQ(q); setIsModalOpen(true); }} className="text-blue-500 hover:bg-blue-50 p-1 rounded"><Icons.Edit /></button>
                                            <button onClick={() => handleDeleteQuestion(q.id)} className="text-red-400 hover:bg-red-50 p-1 rounded"><Icons.Trash /></button>
                                        </div>
                                    </div>
                                ))}
                            </div>
                        </main>

                        <button onClick={() => { setEditingQ(null); setIsModalOpen(true); }} className="fixed bottom-6 right-6 bg-blue-600 text-white h-14 w-14 rounded-full shadow-lg flex items-center justify-center hover:bg-blue-700 z-20">
                            <Icons.Plus />
                        </button>

                        {isModalOpen && (
                            <div className="fixed inset-0 z-50 flex items-center justify-center p-4 modal-overlay backdrop-blur-sm">
                                <div className="bg-white rounded-xl shadow-2xl w-full max-w-sm p-6 animate-in fade-in zoom-in duration-200">
                                    <h2 className="text-lg font-bold mb-4">{editingQ ? 'Editar Pergunta' : 'Nova Pergunta'}</h2>
                                    <form onSubmit={handleSaveQuestion} className="space-y-4">
                                        <div>
                                            <label className="block text-xs font-bold text-slate-500 uppercase mb-1">Setor</label>
                                            <input name="setor" defaultValue={editingQ?.setor || ''} list="setores-list" className="w-full p-2 border rounded-lg text-sm" required placeholder="Ex: ESTOQUE" />
                                            <datalist id="setores-list">{allSetors.map(s => <option key={s} value={s} />)}</datalist>
                                        </div>
                                        <div>
                                            <label className="block text-xs font-bold text-slate-500 uppercase mb-1">Grupo/Categoria</label>
                                            <input name="grupo" defaultValue={editingQ?.grupo || ''} className="w-full p-2 border rounded-lg text-sm" required placeholder="Ex: LIMPEZA" />
                                        </div>
                                        <div>
                                            <label className="block text-xs font-bold text-slate-500 uppercase mb-1">Pergunta</label>
                                            <textarea name="pergunta" defaultValue={editingQ?.pergunta || ''} className="w-full p-2 border rounded-lg text-sm h-24 resize-none" required placeholder="Digite a pergunta..." />
                                        </div>
                                        <div className="flex gap-2 pt-2">
                                            <button type="button" onClick={() => setIsModalOpen(false)} className="flex-1 py-2 border rounded-lg text-slate-600 font-bold">Cancelar</button>
                                            <button type="submit" className="flex-1 py-2 bg-blue-600 text-white rounded-lg font-bold">Salvar</button>
                                        </div>
                                    </form>
                                </div>
                            </div>
                        )}
                    </div>
                );
            }

            if (view === 'inspection') {
                return (
                    <div className="max-w-md mx-auto min-h-screen bg-white flex flex-col animate-fade-in">
                        <header className="sticky top-0 z-10 bg-white border-b border-slate-100 px-6 py-4 flex items-center justify-between shadow-sm">
                            <div><h2 className="text-lg font-bold text-slate-800">{selectedSetor}</h2><p className="text-xs text-slate-500">Resp: {responsavel}</p></div>
                            <div className="text-right"><span className={`text-2xl font-black ${scoreAtual >= 80 ? 'text-green-500' : 'text-red-500'}`}>{scoreAtual}%</span></div>
                        </header>
                        <main className="flex-1 p-6 space-y-6 pb-24 overflow-y-auto">
                            {currentQuestions.map((q, idx) => (
                                <div key={q.id || idx} className="animate-in fade-in slide-in-from-bottom-4 duration-500">
                                    <div className="mb-2"><span className="text-[10px] font-bold text-blue-600 bg-blue-50 px-2 py-1 rounded uppercase">{q.grupo}</span><p className="mt-2 text-slate-800 font-medium leading-snug">{q.pergunta}</p></div>
                                    <div className="flex gap-2">
                                        {['C', 'NC', 'NA'].map(opt => (
                                            <button key={opt} onClick={() => setAnswers(p => ({...p, [q.id]: opt}))} className={`flex-1 py-3 rounded-lg font-bold text-sm transition-colors ${answers[q.id] === opt ? (opt === 'C' ? 'bg-green-500 text-white shadow-lg' : opt === 'NC' ? 'bg-red-500 text-white shadow-lg' : 'bg-slate-600 text-white') : 'bg-slate-100 text-slate-400'}`}>{opt}</button>
                                        ))}
                                    </div>
                                    {answers[q.id] === 'NC' && <input type="text" placeholder="Descreva o problema..." value={obs[q.id] || ''} onChange={(e) => setObs(p => ({...p, [q.id]: e.target.value}))} className="mt-2 w-full text-sm p-3 bg-red-50 border-red-100 rounded-lg text-red-800 focus:border-red-300 outline-none border" autoFocus />}
                                </div>
                            ))}
                        </main>
                        <div className="fixed bottom-0 left-0 right-0 p-4 bg-white border-t border-slate-100 flex gap-3 max-w-md mx-auto">
                            <button onClick={() => setView('dashboard')} className="px-6 py-3 rounded-xl border border-slate-200 font-bold text-slate-500">Cancelar</button>
                            <button onClick={handleSaveInspection} className="flex-1 bg-blue-600 text-white rounded-xl font-bold shadow-lg">Salvar Inspeção</button>
                        </div>
                    </div>
                );
            }
            return null;
        }

        const root = ReactDOM.createRoot(document.getElementById('root'));
        root.render(<App />);
    </script>
